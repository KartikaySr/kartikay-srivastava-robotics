if(RL_USE_QT6)
    find_package(Qt6 COMPONENTS Widgets QUIET)
endif()
if(NOT Qt6_FOUND AND RL_USE_QT5)
    find_package(Qt5 COMPONENTS Widgets QUIET)
endif()
if(Qt6_FOUND OR Qt5_FOUND)
    add_executable(rlRobotics WIN32 Workspace.cpp ${rl_BINARY_DIR}/robotics-library.rc)
    if(APPLE)
        target_sources(rlRobotics PRIVATE robotics.icns)
        set_source_files_properties(robotics.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
        set_target_properties(rlRobotics PROPERTIES MACOSX_BUNDLE ON
            OUTPUT_NAME "Kartikay Srivastava Robotics"
            MACOSX_BUNDLE_BUNDLE_NAME "Kartikay Srivastava Robotics"
            MACOSX_BUNDLE_GUI_IDENTIFIER "com.kartikaysr.robotics"
            MACOSX_BUNDLE_ICON_FILE robotics.icns
            MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION})
    endif()
    rl_add_desktop_resources(rlRobotics)
    if(Qt6_FOUND)
        target_link_libraries(rlRobotics Qt6::Widgets)
    else()
        target_link_libraries(rlRobotics Qt5::Widgets)
    endif()
    foreach(engine fcl bullet ode solid pqp)
        string(TOUPPER ${engine} upper_engine)
        if(RL_BUILD_SG_${upper_engine})
            target_compile_definitions(rlRobotics PRIVATE RL_DESKTOP_ENGINE="${engine}")
            break()
        endif()
    endforeach()
    install(TARGETS rlRobotics RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} BUNDLE DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT demos)
    if(UNIX)
        install(FILES rlRobotics.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications COMPONENT demos)
    endif()
endif()
